var SVC_ALLOWED_ = ["smmxpanel.com", "getmyfollow.com", "smmsun.com", "smmstarpro.com", "smmgen.com",
  "socialpanel.pro", "smmrx.com", "bdsmmpanel.com", "amarboost.com", "growfollows.com"];
function svcAllowed_(p) { return SVC_ALLOWED_.indexOf(String(p)) !== -1; }

var SVC_HEAD_ = ["provider", "service_id", "name", "category", "type", "rate", "min", "max", "refill", "cancel", "dripfeed", "collected_at"];

function ptSvcRoute_(d) {
  var err = adminAuthError_(d.token); if (err) return { ok: false, error: err };
  if (d.provider !== undefined && d.provider !== "" && !svcAllowed_(d.provider)) return { ok: false, error: "no_config" };
  try {
    switch (String(d.action)) {
      case "svc_providers": return svcProviders_();
      case "svc_collect":   return svcCollect_(String(d.provider || ""));
      case "svc_list":      return svcList_(d);
      case "svc_light":     return svcLight_(d);
      case "svc_enrich":    return svcEnrich_(d);
      case "svc_csv":       return svcCsvText_(String(d.provider || ""));
      case "svc_toggle":    return svcToggle_(d);
      case "svc_changes":   return svcChanges_();
      case "svc_auto":      return svcAutoSet_(d);
      case "svc_compare":   return svcCompare_(d);
      case "svc_bulk":      return svcBulk_(d);
      case "svc_tidy_list": return svcTidyList_();
      case "svc_tidy_apply": return svcTidyApply_(d);
      case "svc_platforms":  return svcPlatforms_();
      case "svc_by_platform": return svcByPlatform_(d);
      case "svc_site_list":  return svcSiteListAct_(d);
      case "svc_site_edit":  return svcSiteEdit_(d);
      case "svc_site_hide":  return svcSiteHide_(d);
      case "svc_site_bulk":  return svcSiteBulk_(d);
    }
    return { ok: false, error: "unknown_action" };
  } catch (e) { return { ok: false, error: "svc_failed", message: String(e && e.message || e).substring(0, 200) }; }
}

// ---------- CSV সহায়ক ----------
function svcCsvCell_(v) { var s = String(v == null ? "" : v); return /[",\n\r]/.test(s) ? '"' + s.replace(/"/g, '""') + '"' : s; }
function svcCsvLine_(a) { return a.map(svcCsvCell_).join(","); }
function svcCsvParse_(text) {
  var rows = [], row = [], cell = "", q = false, i, c;
  text = String(text).replace(/^\uFEFF/, "");
  for (i = 0; i < text.length; i++) {
    c = text.charAt(i);
    if (q) { if (c === '"') { if (text.charAt(i + 1) === '"') { cell += '"'; i++; } else q = false; } else cell += c; }
    else if (c === '"') q = true;
    else if (c === ",") { row.push(cell); cell = ""; }
    else if (c === "\n") { row.push(cell); rows.push(row); row = []; cell = ""; }
    else if (c !== "\r") cell += c;
  }
  if (cell !== "" || row.length) { row.push(cell); rows.push(row); }
  return rows;
}
function svcFlag_(v) { return v === true || v === 1 || v === "1" || v === "true" || v === "yes" ? 1 : 0; }

// ---------- ২) লাইভ সংগ্রহ → CSV ----------
function svcCollect_(provider) {
  var cfg = getProviderConfig_()[provider];
  if (!cfg || !cfg.api_key || !cfg.api_url) return { ok: false, error: "no_config" };
  var resp = UrlFetchApp.fetch(cfg.api_url, { method: "post", payload: { key: cfg.api_key, action: "services" }, muteHttpExceptions: true });
  var oldRows = svcRead_(provider);
  var arr; try { arr = JSON.parse(resp.getContentText()); } catch (e) { return { ok: false, error: "bad_json", http: resp.getResponseCode() }; }
  if (!Array.isArray(arr)) return { ok: false, error: "provider_error", message: String(arr && arr.error || "").substring(0, 120) };
  var at = Utilities.formatDate(new Date(), "Asia/Dhaka", "yyyy-MM-dd HH:mm");
  var lines = [svcCsvLine_(SVC_HEAD_)];
  arr.forEach(function (s) {
    if (!s || s.service == null) return;
    lines.push(svcCsvLine_([provider, s.service, String(s.name || "").trim(), String(s.category || "").trim(), s.type || "", s.rate, s.min, s.max,
      s.refill !== undefined ? svcFlag_(s.refill) : (/refill/i.test(s.name || "") ? 1 : 0), svcFlag_(s.cancel), svcFlag_(s.dripfeed), at]));
  });
  var csv = "\uFEFF" + lines.join("\n"), name = svcFileName_(provider), f = svcFindFile_(name);
  if (f) f.setContent(csv); else svcFolder_().createFile(name, csv, MimeType.CSV);
  // ব্যাকআপ: তারিখসহ কপি, শেষ ৫টি রাখা
  var bn = name.replace(".csv", "_" + at.substring(0, 10) + ".csv"), old = svcFindFile_(bn); if (old) old.setTrashed(true);
  svcFolder_().createFile(bn, csv, MimeType.CSV);
  var all = [], it = svcFolder_().searchFiles('title contains "' + name.replace(".csv", "_") + '"');
  while (it.hasNext()) all.push(it.next());
  all.sort(function (a, b) { return b.getName() < a.getName() ? -1 : 1; }); all.slice(5).forEach(function (x) { x.setTrashed(true); });
  var P = PropertiesService.getScriptProperties(), st = svcState_(); st[provider] = { at: at, count: lines.length - 1 };
  P.setProperty("SVC_STATE", JSON.stringify(st));
  try { var sx = svcStatsLoad_(); sx[provider] = svcStatCalc_(arr.filter(function (z) { return z && z.service != null; }).map(function (z) { return { name: z.name, category: z.category }; })); svcStatsSave_(sx); } catch (e) {}
  var df = null;
  if (oldRows) { try { df = svcDiff_(oldRows, arr); var ch = svcChgLoad_(); ch[provider] = { at: at, c: df.c, items: df.items }; svcChgSave_(ch); } catch (e) {} }
  return { ok: true, provider: provider, count: lines.length - 1, collected_at: at, changes: df ? df.c : null };
}

function svcRead_(provider) {
  var f = svcFindFile_(svcFileName_(provider)); if (!f) return null;
  var rows = svcCsvParse_(f.getBlob().getDataAsString("UTF-8")), h = rows.shift() || [], out = [];
  rows.forEach(function (r) { if (r.length < 3) return; var o = {}; h.forEach(function (k, i) { o[k] = r[i]; }); out.push(o); });
  return out;
}
function svcCsvText_(provider) { var f = svcFindFile_(svcFileName_(provider)); return f ? { ok: true, csv: f.getBlob().getDataAsString("UTF-8"), name: svcFileName_(provider) } : { ok: false, error: "not_collected" }; }

function svcSuggest_(rateUsd) {   // সাইটে বিক্রয়মূল্য ≈ প্রভাইডার রেট + সাইটের মার্কআপ নিয়ম (শুধু দেখানোর জন্য; সাইট নিজেই মার্কআপ বসায়)

// ---------- ৩) তালিকা ----------
function svcList_(d) {
  var provider = String(d.provider || ""), rows = svcRead_(provider);
  if (!rows) return { ok: false, error: "not_collected" };
  var meta = getServiceMeta_(), live = {}, hidden = {}, cats = {}, siteCats = {};
  meta.custom.forEach(function (c) { if (c.category) siteCats[c.category] = 1; if (!c.ov) live[String(c.code)] = c; });
  meta.hidden.forEach(function (c) { hidden[String(c)] = 1; });
  var tiers = null, fx = svcFx_(); try { tiers = ptTiersLoad_(); } catch (e) {}
  var pf = String(d.platform || ""), q = String(d.q || "").toLowerCase().trim(), cat = String(d.category || ""), flt = String(d.state || "all");
  rows.forEach(function (r) { cats[r.category] = (cats[r.category] || 0) + 1; r.state = "off"; });
  var out = rows.filter(function (r) {
    var c = live[r.service_id];
    if (c && (!c.provider || c.provider === provider)) r.state = hidden[r.service_id] ? "off" : "on";
    else if (c) r.state = "conflict", r.conflict_with = c.provider;
    if (cat && r.category !== cat) return false;
    if (pf && svcPlatOf_(r) !== pf) return false;
    if (flt !== "all" && r.state !== flt) return false;
    if (q && (r.name + " " + r.service_id + " " + r.category).toLowerCase().indexOf(q) < 0) return false;
    return true;
  });
  var off = Math.max(0, Number(d.offset) || 0), lim = Math.min(200, Number(d.limit) || 100), page = out.slice(off, off + lim);
  var spd = svcSpd_(), cmap = svcCatMap_();
  page.forEach(function (r) { r.suggest = svcSuggest_(Number(r.rate), tiers, fx); r.spd = spd[r.service_id] || null; r.tidy = svcTidy_(r, cmap); });
  return { ok: true, total: out.length, rows: page, categories: cats, site_categories: Object.keys(siteCats).sort(), fx: fx, tiers: !!tiers, platforms: svcProvPlats_(provider) };
}

// ---------- ৪) অন/অফ ----------
function svcToggle_(d) {
  var provider = String(d.provider || ""), code = String(d.code || "").trim(), meta = getServiceMeta_();
  if (!provider || !code) return { ok: false, error: "code_required" };
  var prev = null; meta.custom.forEach(function (c) { if (String(c.code) === code && !c.ov) prev = c; });
  if (d.on) {
    if (prev && prev.provider && prev.provider !== provider) return { ok: false, error: "code_conflict", with: prev.provider,
      message: "এই আইডি (" + code + ") সাইটে আগেই " + prev.provider + "-এর সার্ভিস হিসেবে আছে" };
    var row = (svcRead_(provider) || []).filter(function (r) { return r.service_id === code; })[0];
    if (!row) return { ok: false, error: "not_in_csv" };
    var cat = String(d.category || "").trim(), plat = String(d.platform || "other").toLowerCase().trim(), usd = Number(row.rate);   // সাইটে pricePerK = প্রভাইডারের খরচ; মার্কআপ সাইট নিজেই বসায়
    if (!cat) return { ok: false, error: "category_required" };
    if (!(usd > 0)) return { ok: false, error: "price_required" };
    var rm = meta.removed[code]; if (rm && rm.provider && rm.provider !== provider) rm = null;
    if (rm && !d.force) return { ok: false, error: "previously_removed", removed_info: rm, message: "এই সার্ভিস আগে সাইট থেকে বাদ দেওয়া হয়েছিল" };
    var entry = { code: code, name: String(d.name || row.name).trim().substring(0, 120), platform: plat, category: cat, tags: [], pricePerK: "$" + usd,
      min: Number(row.min) || 10, max: Number(row.max) || 10000, popular: false, provider: provider };
    if (row.refill === "1") entry.refill = "yes";
    meta.custom = meta.custom.filter(function (c) { return String(c.code) !== code; }); meta.custom.push(entry);
    meta.hidden = meta.hidden.filter(function (c) { return String(c) !== code; }); delete meta.removed[code];
    try { svcCatLearn_(provider, row.category, cat); } catch (e) {}
    saveServiceMeta_(meta); logServiceEvent_(code, "add", entry, "service_studio");
    return { ok: true, state: "on", code: code };
  }
  if (!prev) return { ok: true, state: "off", code: code };
  if (meta.hidden.indexOf(code) === -1) meta.hidden.push(code);
  markServiceRemoved_(meta, code, "hide"); saveServiceMeta_(meta); logServiceEvent_(code, "hide", {}, "service_studio");
  return { ok: true, state: "off", code: code };
}

function svcDiff_(oldRows, arr) {
  var o = {}, n = {}, items = [], cnt = { up: 0, down: 0, rm: 0, nw: 0 }, per = { up: 0, down: 0, rm: 0, nw: 0 };
  function add(k, t, it) { cnt[k]++; if (per[k] < 100) { per[k]++; it.t = t; items.push(it); } }
  (oldRows || []).forEach(function (r) { o[r.service_id] = r; });
  arr.forEach(function (s) { if (s && s.service != null) n[String(s.service)] = s; });
  Object.keys(n).forEach(function (id) {
    var s = n[id], r = o[id], nm = String(s.name || "").substring(0, 70);
    if (!r) return add("nw", "new", { id: id, nm: nm, n: Number(s.rate) });
    var a = Number(r.rate), b = Number(s.rate);
    if (a > 0 && b > 0 && Math.abs(a - b) / a > 0.0001) { var k = b > a ? "up" : "down"; add(k, k, { id: id, nm: nm, o: a, n: b }); }
  });
  Object.keys(o).forEach(function (id) { if (!n[id]) add("rm", "rm", { id: id, nm: String(o[id].name || "").substring(0, 70), o: Number(o[id].rate) }); });
  return { c: cnt, items: items };
}

function svcChanges_() {
  var ch = svcChgLoad_(), meta = getServiceMeta_(), live = {}, hid = {}, out = [], ord = { rm: 0, up: 1, down: 2, "new": 3 };
  meta.custom.forEach(function (c) { if (!c.ov) live[String(c.code)] = c.provider || ""; });
  meta.hidden.forEach(function (c) { hid[String(c)] = 1; });
  SVC_ALLOWED_.forEach(function (p) {
    var e = ch[p]; if (!e) return;
    var lr = 0, lu = 0;
    e.items.forEach(function (i) {
      i.live = (i.id in live) && (live[i.id] === p || live[i.id] === "") && !hid[i.id] ? 1 : 0;
      if (i.live && i.t === "rm") lr++; if (i.live && i.t === "up") lu++;
    });
    e.items.sort(function (a, b) { return (b.live - a.live) || (ord[a.t] - ord[b.t]); });
    out.push({ provider: p, at: e.at, c: e.c, live_rm: lr, live_up: lu, items: e.items.slice(0, 60) });
  });
  return { ok: true, providers: out, auto: svcAutoInfo_() };
}

// ---------- সাপ্তাহিক অটো-সংগ্রহ: রোজ ভোর ৩টায় চলে, ৭ দিনের পুরনো সর্বোচ্চ ৩টা প্রভাইডার সংগ্রহ করে ----------
function svcTrig_(on) {
  ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === "svcAutoRun") ScriptApp.deleteTrigger(t); });
  if (on) ScriptApp.newTrigger("svcAutoRun").timeBased().everyDays(1).atHour(3).create();
}
function svcAutoInstall() { svcTrig_(true); }   // দরকার হলে এডিটর থেকে একবার চালান (অনুমতির জন্য)
function svcAutoInfo_() {
  var on = false;
  try { ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === "svcAutoRun") on = true; }); } catch (e) { on = null; }
  return { on: on, last: PropertiesService.getScriptProperties().getProperty("SVC_AUTO_LAST") || "" };
}
function svcAutoSet_(d) {
  try { svcTrig_(!!d.on); }
  catch (e) { return { ok: false, error: "trigger_failed", message: "অনুমতি লাগবে: Apps Script এডিটরে svcAutoInstall ফাংশনটা একবার Run করুন" }; }
  return { ok: true, auto: svcAutoInfo_() };
}
function svcAutoRun() {
  var st = svcState_(), t0 = Date.now(), due = [], done = [];
  SVC_ALLOWED_.forEach(function (p) {
    var at = st[p] && st[p].at, age = at ? (t0 - new Date(String(at).replace(" ", "T") + "+06:00").getTime()) / 864e5 : 99;
    if (!(age < 7)) due.push({ p: p, age: age });
  });
  due.sort(function (a, b) { return b.age - a.age; });
  due.slice(0, 3).forEach(function (x) {
    if (Date.now() - t0 > 240000) return;
    try { if (svcCollect_(x.p).ok) done.push(x.p); } catch (e) {}
  });
  PropertiesService.getScriptProperties().setProperty("SVC_AUTO_LAST", Utilities.formatDate(new Date(), "Asia/Dhaka", "yyyy-MM-dd HH:mm") + " · " + done.length + "/" + due.length);
  if (!done.length) return;
  try {
    var a = 0; svcChanges_().providers.forEach(function (x) { if (done.indexOf(x.provider) >= 0) a += x.live_rm + x.live_up; });
    if (a && OWNER_EMAIL) MailApp.sendEmail(OWNER_EMAIL, "সার্ভিস স্টুডিও: চালু সার্ভিসে পরিবর্তন", a + "টি চালু সার্ভিসে দাম বেড়েছে বা প্রভাইডার থেকে বাদ পড়েছে। স্টুডিও > পরিবর্তন দেখুন।");
  } catch (e) {}
}

// ================= দাম তুলনা (১০ প্রভাইডারে একসাথে খোঁজা, সস্তা আগে) =================
function svcCompare_(d) {
  var q = String(d.q || "").toLowerCase().trim().split(/\s+/).filter(Boolean);
  if (q.join("").length < 2) return { ok: true, rows: [], total: 0 };
  var cmap = svcCatMap_(), meta = getServiceMeta_(), live = {}, hid = {}, sc = {}, spd = svcSpd_(), fx = svcFx_(), tiers = null, all = [];
  meta.custom.forEach(function (c) { if (c.category) sc[c.category] = 1; if (!c.ov) live[String(c.code)] = c; });
  meta.hidden.forEach(function (c) { hid[String(c)] = 1; });
  try { tiers = ptTiersLoad_(); } catch (e) {}
  SVC_ALLOWED_.forEach(function (p) {
    (svcRead_(p) || []).forEach(function (r) {
      var h = (r.name + " " + r.category).toLowerCase();
      for (var i = 0; i < q.length; i++) if (h.indexOf(q[i]) < 0) return;
      if (d.refill && r.refill !== "1") return;
      all.push(r);
    });
  });
  all.sort(function (a, b) { return Number(a.rate) - Number(b.rate); });

function svcBulk_(d) {
  var provider = String(d.provider || ""), codes = (d.codes || []).map(String).slice(0, 100), meta = getServiceMeta_(), by = {}, ok = 0, skip = 0, tiers = null, fx = svcFx_();
  (svcRead_(provider) || []).forEach(function (r) { by[r.service_id] = r; });
  try { tiers = ptTiersLoad_(); } catch (e) {}
  var cat = String(d.category || "").trim(), mk = (d.markup === "" || d.markup == null) ? null : Number(d.markup);
  var cmap = svcCatMap_();
  codes.forEach(function (code) {
    var prev = null; meta.custom.forEach(function (c) { if (String(c.code) === code && !c.ov) prev = c; });
    if (prev && prev.provider && prev.provider !== provider) { skip++; return; }
    if (!d.on) {
      if (!prev) { skip++; return; }
      if (meta.hidden.indexOf(code) === -1) meta.hidden.push(code);
      markServiceRemoved_(meta, code, "hide"); logServiceEvent_(code, "hide", {}, "service_studio"); ok++; return;
    }
    var row = by[code]; if (!row) { skip++; return; }
    var rm = meta.removed[code]; if (rm && rm.provider && rm.provider !== provider) rm = null;
    if (rm && !d.force) { skip++; return; }
    var usd = Math.round(Number(row.rate) * 1e6) / 1e6;   // প্রভাইডারের রেটই বসবে; মার্কআপ সাইট নিজে দেবে if (!(usd > 0)) { skip++; return; }
    var t = svcTidy_(row, cmap);
    var entry = { code: code, name: t.name.substring(0, 120), platform: t.plat, category: cat || t.cat, tags: [], pricePerK: "$" + usd,
      min: Number(row.min) || 10, max: Number(row.max) || 10000, popular: false, provider: provider };
    if (row.refill === "1") entry.refill = "yes";
    meta.custom = meta.custom.filter(function (c) { return String(c.code) !== code; }); meta.custom.push(entry);
    meta.hidden = meta.hidden.filter(function (c) { return String(c) !== code; }); delete meta.removed[code];
    logServiceEvent_(code, "add", entry, "service_studio"); ok++;
  });
  if (ok) saveServiceMeta_(meta);
  return { ok: true, done: ok, skipped: skip };
}

function svcPick_(list, text) {
  for (var i = 0; i < list.length; i++) { var re = list[i][list[i].length - 1]; if (re.test(text)) return list[i]; }
  return null;
}
function svcClean_(t, keep) {
  var s = String(t || "").replace(/[\u{1F000}-\u{1FFFF}\u{2190}-\u{2BFF}\uFE0F\u200D\u20E3]/gu, " ");
  s = s.replace(/[\[\(\{]([^\]\)\}]*)[\]\)\}]/g, function (m, i) {
    return keep && /\d/.test(i) && /min|hour|hr|day|%|\d\s?[kKmMdDhH]\b/i.test(i) && !/refill|drop/i.test(i) ? " (" + i.trim() + ") " : " ";
  });
  return s.replace(/[|•~*_=#>]+/g, " ").replace(/\s+/g, " ").replace(/^[\s\-–—:.]+|[\s\-–—:.]+$/g, "");
}
function svcTags_(t, refillFlag) {
  var o = [], m;
  if (/\bhq\b|high.?quality|premium/i.test(t)) o.push("HQ"); else if (/\breal\b|organic/i.test(t)) o.push("Real");
  m = t.match(/\b(usa|uk|india|indian|bangladesh|bd|turkey|brazil|arab|global|worldwide)\b/i);
  if (m) o.push(m[1].length <= 3 ? m[1].toUpperCase() : m[1].charAt(0).toUpperCase() + m[1].slice(1).toLowerCase());
  if (/non.?drop|no.?drop|\bnd\b/i.test(t)) o.push("Non-Drop");
  var tr = t.replace(/\b(no|non|without)[\s\-]?refill\b/ig, " ");
  if (/refill|♻|lifetime|\blt\b|\br\d{2,3}\b/i.test(tr) || refillFlag) {
    m = tr.match(/(\d{2,3})\s*(?:d|days?)\b|\br(\d{2,3})\b/i);
    o.push((m ? (m[1] || m[2]) + "D " : (/lifetime|\blt\b/i.test(tr) ? "Lifetime " : "")) + "Refill");
  }
  m = t.match(/(\d+)\s*(minutes?|mins?|hours?|hrs?)\b/i);
  if (m) o.push(m[1] + (/^m/i.test(m[2]) ? " Min" : " Hr"));
  if (/instant|super.?fast|\bfast\b/i.test(t)) o.push("Fast");
  return o.slice(0, 4);
}

function svcTidy_(r, map) {
  var c0 = String(r.category || ""), n0 = String(r.name || ""), all = c0 + " " + n0;
  var P = svcPick_(SVC_PLAT_, c0) || svcPick_(SVC_PLAT_, n0), T = svcPick_(SVC_TYPE_, c0) || svcPick_(SVC_TYPE_, n0);
  var mapped = map && map[(r.provider || "") + "|" + c0], base = (P && T) ? P[0] + " " + T[0] : "";
  var cat = mapped || base || svcClean_(c0) || "Other", name;
  if (base) { var tg = svcTags_(all, String(r.refill) === "1"); name = base + (tg.length ? " – " + tg.join(" · ") : ""); }
  else name = svcClean_(n0, true) || n0;
  return { name: name, cat: cat, plat: P ? P[1] : "other" };
}

    if (d.tidy) { var t = svcTidy_({ provider: "", name: r.name, category: r.category, refill: 0 }, cmap); name = t.name; ct = t.cat; }
    if (cat) ct = cat;
    var c = cust[code]; if (!c) { c = { code: code, ov: 1 }; meta.custom.push(c); cust[code] = c; }
    c.name = name.substring(0, 120); c.category = ct.substring(0, 80); if (plat) c.platform = plat;
    n++;
  });
  if (n) saveServiceMeta_(meta);
  return { ok: true, done: n };
}

// ---------- একটা প্রভাইডারের প্ল্যাটফর্ম-তালিকা (লোগো-টাইলের জন্য) ----------
function svcProvPlats_(provider) {
  var st = svcStatsLoad_(), c = st[provider];
  if (!c) { var rows = svcRead_(provider); if (rows) { c = st[provider] = svcStatCalc_(rows); try { svcStatsSave_(st); } catch (e) {} } }
  return Object.keys(c || {}).map(function (k) { return { label: k, n: c[k] }; }).sort(function (a, b) { return b.n - a.n; });
}

// ================= ফোনে-CSV মোডের জন্য হালকা অ্যাকশন (CSV পড়ে না — তাই দ্রুত) =================
// চালু সার্ভিসের আইডি-তালিকা + জট। অ্যাপ CSV নিজের ফোন থেকে পড়ে, শুধু অন/অফ অবস্থা এখান থেকে নেয়।
function svcLight_(d) {
  var provider = String(d.provider || ""), meta = getServiceMeta_(), live = {}, hidden = {}, siteCats = {}, on = [], cf = {};
  meta.custom.forEach(function (c) { if (c.category) siteCats[c.category] = 1; if (!c.ov) live[String(c.code)] = c; });
  meta.hidden.forEach(function (c) { hidden[String(c)] = 1; });
  Object.keys(live).forEach(function (id) {
    var c = live[id];
    if (!c.provider || c.provider === provider) { if (!hidden[id]) on.push(id); } else cf[id] = c.provider;
  });
  return { ok: true, on: on, cf: cf, site_categories: Object.keys(siteCats).sort(), fx: svcFx_() };
}
// দেখানো পাতার (সর্বোচ্চ ২০০টি) সারির দাম-প্রস্তাব, স্পিড, সাজানো নাম। সারি অ্যাপ পাঠায়, তাই CSV পড়তে হয় না।
function svcEnrich_(d) {
  var provider = String(d.provider || ""), rows = (d.rows || []).slice(0, 200), spd = svcSpd_(), cmap = svcCatMap_(), items = {};
  rows.forEach(function (r) {
    var id = String(r.service_id);
    items[id] = { suggest: svcSuggest_(Number(r.rate)), spd: spd[id] || null,
      tidy: svcTidy_({ provider: provider, name: r.name, category: r.category, refill: r.refill }, cmap) };
  });
  return { ok: true, items: items };
}
