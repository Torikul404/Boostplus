function svcDiff_(oldRows, arr) {

  arr.forEach(function (s) { if (s && s.service != null) n[String(s.service)] = s; });
  Object.keys(n).forEach(function (id) {
    var s = n[id], r = o[id], nm = svcCut_(String(s.name || ""), 70);
    if (!r) return add("nw", "new", { id: id, nm: nm, n: Number(s.rate) });

/** ইমোজি (surrogate pair) মাঝখানে না ভেঙে n অক্ষরে কাটে — আগে নাম/ক্যাটেগরি কাটলে "??" ভাঙা চিহ্ন হতো। */
function svcCut_(s, n) {
  s = String(s == null ? "" : s); if (s.length <= n) return s;
  var t = s.substring(0, n), c = t.charCodeAt(n - 1);
  if (c >= 0xD800 && c <= 0xDBFF) t = t.substring(0, n - 1);
  return t;
}
